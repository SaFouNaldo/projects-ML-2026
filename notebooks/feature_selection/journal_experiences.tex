\section{Conventions}
\begin{itemize}
  \item Decision metric: \textbf{F1} (positive class = MICHD).
  \item Train prevalence: $28\,975 / 328\,135 \approx 8.83\,\%$.
  \item Reference pipeline: \textbf{base} (54 raw $\to$ preprocess $\sim$103 cols)
        + 7 extras \texttt{NO\_DUP\_EXTRAS} $\to$ $\sim$110 columns
        (\texttt{pipeline.py}).
  \item Reliable evaluation (since Oct.\ 2026): \textbf{stratified 5-fold OOF + IRLS},
        F1 threshold on pooled OOF scores.
  \item F1 noise (OOF bootstrap): $\sigma \approx 0.002$.
        A variant is kept only if $\Delta\mathrm{F1} \gtrsim 0.005$
        ($\approx 2\sigma$), unless stated otherwise.
  \item Older measurements = multi-seed 80/20 holdout + GD logistic (noisier;
        orders of magnitude only).
\end{itemize}

\section{Details --- top-$K$ interactions \& GD tuning (6 Oct.\ 2026)}
Interactions = products of the $K$ columns with the largest $|w|$ (IRLS), excluding the bias.

\begin{tabular}{@{}lccc@{}}

Top-12 pairs (+66) & 0.4264 & $+0.0036$ & 0.8600 \\
\bottomrule
\end{tabular}
\end{center}

\noindent\textit{GD (course code)} grid $\gamma\in\{0.05,0.1,0.5,1\}$,
\texttt{max\_iters}$\in\{1000,3000,5000\}$ then checked on the 80\,\% train split:
best config seen: $\gamma=0.5$, $5000$ iters, top-12 inter,
holdout F1 $\approx 0.432$ (a more optimistic protocol than OOF).
\textbf{Decision (6 Oct.): integrated into the frozen pipeline}
(\texttt{interactions.TopKPairInteractions}, $K=12$).

\section{Next steps}
\begin{enumerate}
  \item Add 1--3 non-redundant OUT columns (e.g.\ vaccines, \texttt{HAVARTH3}) + OOF.
  \item Bins for \texttt{GENHLTH} / BMI / health days (optional).
